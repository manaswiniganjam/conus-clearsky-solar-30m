$stations = @(
@{code="bon"; dir="Bondville_IL"},
@{code="fpk"; dir="Fort_Peck_MT"},
@{code="gwn"; dir="Goodwin_Creek_MS"},
@{code="tbl"; dir="Table_Mountain_CO"},
@{code="dra"; dir="Desert_Rock_NV"},
@{code="psu"; dir="Penn_State_PA"},
@{code="sxf"; dir="Sioux_Falls_SD"}
)

$doys = @("015","045","074","105","135","166","196","227","258","288","319","349")
$years = 2015..2024
$baseUrl = "https://gml.noaa.gov/aftp/data/radiation/surfrad"
$outBase = "C:\Users\mgvhy\surfrad_data"

foreach ($st in $stations) {
foreach ($year in $years) {
$yy = $year.ToString().Substring(2)
$outDir = "$outBase\$($st.code)\$year"
New-Item -ItemType Directory -Force -Path $outDir | Out-Null
foreach ($doy in $doys) {
$url = "$baseUrl/$($st.dir)/$year/$($st.code)$yy$doy.dat"
$out = "$outDir\$($st.code)$yy$doy.dat"
try {
Invoke-WebRequest -Uri $url -OutFile $out -ErrorAction Stop
Write-Host "OK: $($st.code) $year $doy"
} catch {
Write-Host "MISSING: $($st.code) $year $doy"
}
}
}
}
Write-Host "Done"
